REGRA RN-01 • Validação Oficial de CPF (Módulo 11)

A identificação do cliente segue o algoritmo oficial da Receita Federal. O Módulo onze calcula os dois dígitos verificadores e bloqueia qualquer CPF inválido ou duplicado.

src/utils/cpf-validator.ts (Algoritmo Módulo 11) RN-01 HOMOLOGADA 
export function validarCPF(cpf: string): boolean {
  const limpo = cpf.replace(/\D/g, '');
  if (limpo.length !== 11 || /^(\d)\1{10}$/.test(limpo)) return false;

  // Cálculo do 1º Dígito (Pesos 10 a 2)
  let d1 = 11 - (somarPesos(limpo, 9, 10) % 11);
  if (d1 >= 10) d1 = 0;
  if (d1 !== Number(limpo[9])) return false; // Bloqueio imediato de CPF falso

  // Cálculo do 2º Dígito (Pesos 11 a 2)
  let d2 = 11 - (somarPesos(limpo, 10, 11) % 11);
  if (d2 >= 10) d2 = 0;
  return d2 === Number(limpo[10]); // CPF 100% Válido na Receita
}
