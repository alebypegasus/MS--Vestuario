export function validarCPF(cpf: string): boolean {
  const limpo = cpf.replace(/\D/g, '');
  if (limpo.length !== 11 || /^(\d)\1{10}$/.test(limpo)) return false;

  // Cálculo do 1º DV (Pesos 10 a 2)
  let soma = 0;
  for (let i = 0; i < 9; i++) soma += parseInt(limpo[i]) * (10 - i);
  let dv1 = (11 - (soma % 11)) >= 10 ? 0 : (11 - (soma % 11));
  if (dv1 !== parseInt(limpo[9])) return false;

  // Cálculo do 2º DV (Pesos 11 a 2)
  soma = 0;
  for (let i = 0; i < 10; i++) soma += parseInt(limpo[i]) * (11 - i);
  let dv2 = (11 - (soma % 11)) >= 10 ? 0 : (11 - (soma % 11));
  return dv2 === parseInt(limpo[10]); // 100% Homologado
}
